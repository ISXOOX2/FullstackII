/** Elimina del objeto los campos sin error (''), dejando solo los que fallaron. */
function soloErrores(errores) {
  const resultado = {};
  Object.keys(errores).forEach((campo) => {
    if (errores[campo]) resultado[campo] = errores[campo];
  });
  return resultado;
}
